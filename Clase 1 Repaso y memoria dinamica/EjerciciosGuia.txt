typedef struct nodo {
    char palo;
    int nro;
    struct nodo *sig;
} nodo;

typedef nodo *Tlista;

int main() {

    Tlista n1, n2;


    n1 = (Tlista) malloc(sizeof(nodo));
    n2 = (Tlista) malloc(sizeof(nodo));

    if (n1 == NULL || n2==NULL) {
        printf("Error de memoria\n");

    } else{


    n1->nro = 7;
    n1->palo = 'C';

    n2->nro = 12;
    n2->palo = 'E';

    // Enlazar nodos
    n1->sig = n2;     // n1 apunta a n2
    n2->sig = NULL;   // último nodo apunta a NULL


    printf("Nodo 1: %d %c\n", n1->nro, n1->palo);
    printf("Nodo 2: %d %c\n", n1->sig->nro, n1->sig->palo);


    free(n2);
    free(n1);

    }
    return 0;
}
